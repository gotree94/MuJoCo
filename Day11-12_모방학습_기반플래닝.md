## 1. 모방학습 이론

### 1.1 모방학습 개요

```
지도학습 기반 모방학습:
Expert demonstrations D = {(s_i, a_i)} → Policy π(a|s)

강화학습과의 차이:
- 보상 함수 불필요
- Expert 데이터로 직접 학습
- 빠른 학습 속도
- 분포 이동 (Distribution Shift) 문제
```

### 1.2 Behavior Cloning (BC)

```
목표: Expert의 행동을 직접 모방
손실 함수: L = E_{(s,a)~D} [loss(π(s), a)]

손실 함수 종류:
- MSE: ||π(s) - a||² (연속 행동)
- Cross Entropy: -Σ a log π(s) (이산 행동)
```

### 1.3 DAgger (Dataset Aggregation)

```
문제: BC의 분포 이동
해결: Expert의 라벨링을 반복적으로 추가

알고리즘:
1. 초기 expert 데이터 D₀ 수집
2. for k = 1 to K:
   a. 현재 policy π_k로 환경 실행
   b. 실행 중 상태 s_i 관찰
   c. Expert에게 s_i에 대한 올바른 행동 a_i 요청
   d. D_k = D_{k-1} ∪ {(s_i, a_i)}
   e. D_k로 새 policy π_{k+1} 학습
```

## 2. Demonstration 데이터 수집

### 2.1 MuJoCo에서 Expert 정책 구현

```python
import mujoco
import numpy as np
import h5py
import os

class ExpertPolicy:
    def __init__(self, model, data):
        self.model = model
        self.data = data
    
    def get_action(self, state):
        """Expert 행동 결정 (PD 제어 기반)"""
        # 상태에서 목표 설정
        target_pos = np.array([0.3, 0.0, 0.5])  # 목표 위치
        
        # 간단한 IK (Inverse Kinematics)
        current_pos = self.data.site_xpos[0]  # 엔드이펙터 위치
        error = target_pos - current_pos
        
        # PD 제어
        kp = 50
        kd = 10
        
        action = kp * error - kd * self.data.qvel[:self.model.nu]
        
        return action

class DemonstrationCollector:
    def __init__(self, model_path, save_dir='demonstrations'):
        self.model = mujoco.MjModel.from_xml_path(model_path)
        self.data = mujoco.MjData(self.model)
        self.expert = ExpertPolicy(self.model, self.data)
        self.save_dir = save_dir
        
        os.makedirs(save_dir, exist_ok=True)
    
    def collect_demonstration(self, demo_id, duration=5.0, dt=0.002):
        """단일 demonstration 수집"""
        states = []
        actions = []
        
        steps = int(duration / dt)
        
        for i in range(steps):
            # 현재 상태 저장
            state = np.concatenate([
                self.data.qpos,
                self.data.qvel
            ])
            
            # Expert 행동
            action = self.expert.get_action(state)
            
            states.append(state)
            actions.append(action)
            
            # 액추레이터에 적용
            self.data.ctrl[:self.model.nu] = action
            
            # 시뮬레이션 스텝
            mujoco.mj_step(self.model, self.data)
        
        return np.array(states), np.array(actions)
    
    def collect_multiple_demonstrations(self, num_demos=100, duration=5.0):
        """여러 demonstration 수집"""
        all_states = []
        all_actions = []
        
        for demo_id in range(num_demos):
            print(f"Collecting demonstration {demo_id + 1}/{num_demos}")
            
            # 초기 상태 랜덤화
            self.reset_random()
            
            states, actions = self.collect_demonstration(demo_id, duration)
            all_states.append(states)
            all_actions.append(actions)
        
        # HDF5 파일로 저장
        self.save_demonstrations(all_states, all_actions)
        
        return all_states, all_actions
    
    def reset_random(self):
        """랜덤 초기 상태"""
        for i in range(self.model.njnt):
            joint_range = self.model.jnt_range[i]
            self.data.qpos[i] = np.random.uniform(
                joint_range[0], joint_range[1]
            )
        self.data.qvel[:] = 0
    
    def save_demonstrations(self, states_list, actions_list, filename='demos.hdf5'):
        """HDF5로 저장"""
        filepath = os.path.join(self.save_dir, filename)
        
        with h5py.File(filepath, 'w') as f:
            # 전체 데이터 저장
            all_states = np.concatenate(states_list, axis=0)
            all_actions = np.concatenate(actions_list, axis=0)
            
            f.create_dataset('states', data=all_states)
            f.create_dataset('actions', data=all_actions)
            
            # demonstration별 인덱스 저장
            indices = [0]
            for states in states_list:
                indices.append(indices[-1] + len(states))
            f.create_dataset('indices', data=indices)
            
            # 메타데이터
            f.attrs['num_demos'] = len(states_list)
            f.attrs['state_dim'] = all_states.shape[1]
            f.attrs['action_dim'] = all_actions.shape[1]
        
        print(f"Saved {len(states_list)} demonstrations to {filepath}")

# 사용 예시
collector = DemonstrationCollector('robot.xml')
collector.collect_multiple_demonstrations(num_demos=50, duration=5.0)
```

## 3. Behavior Cloning 구현

### 3.1 데이터 로더

```python
import torch
from torch.utils.data import Dataset, DataLoader
import h5py
import numpy as np

class DemonstrationDataset(Dataset):
    def __init__(self, hdf5_path, transform=None):
        with h5py.File(hdf5_path, 'r') as f:
            self.states = f['states'][:]
            self.actions = f['actions'][:]
        
        self.transform = transform
        
        # 정규화
        self.state_mean = self.states.mean(axis=0)
        self.state_std = self.states.std(axis=0) + 1e-8
        
        self.action_mean = self.actions.mean(axis=0)
        self.action_std = self.actions.std(axis=0) + 1e-8
    
    def __len__(self):
        return len(self.states)
    
    def __getitem__(self, idx):
        state = self.states[idx]
        action = self.actions[idx]
        
        # 정규화
        state = (state - self.state_mean) / self.state_std
        action = (action - self.action_mean) / self.action_std
        
        if self.transform:
            state = self.transform(state)
        
        return torch.FloatTensor(state), torch.FloatTensor(action)
    
    def denormalize_action(self, action_norm):
        """정규화된 행동 복원"""
        return action_norm * self.action_std + self.action_mean
```

### 3.2 정책 네트워크

```python
import torch
import torch.nn as nn

class BehaviorCloningPolicy(nn.Module):
    def __init__(self, state_dim, action_dim, hidden_dims=[256, 256, 128]):
        super().__init__()
        
        layers = []
        prev_dim = state_dim
        
        for hidden_dim in hidden_dims:
            layers.extend([
                nn.Linear(prev_dim, hidden_dim),
                nn.ReLU(),
                nn.LayerNorm(hidden_dim),
                nn.Dropout(0.1)
            ])
            prev_dim = hidden_dim
        
        layers.append(nn.Linear(prev_dim, action_dim))
        
        self.network = nn.Sequential(*layers)
    
    def forward(self, state):
        return self.network(state)
    
    def predict(self, state):
        """단일 상태 예측"""
        with torch.no_grad():
            if isinstance(state, np.ndarray):
                state = torch.FloatTensor(state)
            if state.dim() == 1:
                state = state.unsqueeze(0)
            return self.forward(state).squeeze(0).numpy()
```

### 3.3 학습 루프

```python
import torch
import torch.nn as nn
from torch.optim import Adam
from torch.optim.lr_scheduler import CosineAnnealingLR
import numpy as np
from tqdm import tqdm

class BCTrainer:
    def __init__(self, policy, dataset, device='cuda'):
        self.policy = policy.to(device)
        self.dataset = dataset
        self.device = device
        
        self.optimizer = Adam(policy.parameters(), lr=1e-3, weight_decay=1e-5)
        self.scheduler = CosineAnnealingLR(self.optimizer, T_max=100)
        self.criterion = nn.MSELoss()
    
    def train(self, num_epochs=100, batch_size=64, val_split=0.1):
        """학습 메인 루프"""
        # 데이터 분할
        val_size = int(len(self.dataset) * val_split)
        train_size = len(self.dataset) - val_size
        
        train_dataset, val_dataset = torch.utils.data.random_split(
            self.dataset, [train_size, val_size]
        )
        
        train_loader = DataLoader(
            train_dataset, batch_size=batch_size, shuffle=True
        )
        val_loader = DataLoader(
            val_dataset, batch_size=batch_size, shuffle=False
        )
        
        # 학습 기록
        train_losses = []
        val_losses = []
        best_val_loss = float('inf')
        
        for epoch in range(num_epochs):
            # 학습
            self.policy.train()
            train_loss = self.train_epoch(train_loader)
            train_losses.append(train_loss)
            
            # 검증
            self.policy.eval()
            val_loss = self.validate(val_loader)
            val_losses.append(val_loss)
            
            # 스케줄러 업데이트
            self.scheduler.step()
            
            # 모델 저장
            if val_loss < best_val_loss:
                best_val_loss = val_loss
                torch.save(self.policy.state_dict(), 'best_bc_policy.pth')
            
            # 로그
            if (epoch + 1) % 10 == 0:
                print(f"Epoch {epoch+1}/{num_epochs}")
                print(f"  Train Loss: {train_loss:.6f}")
                print(f"  Val Loss: {val_loss:.6f}")
        
        return train_losses, val_losses
    
    def train_epoch(self, loader):
        """에포크 학습"""
        total_loss = 0
        
        for states, actions in loader:
            states = states.to(self.device)
            actions = actions.to(self.device)
            
            # 예측
            pred_actions = self.policy(states)
            
            # 손실 계산
            loss = self.criterion(pred_actions, actions)
            
            # 역전파
            self.optimizer.zero_grad()
            loss.backward()
            
            # 그래디언트 클리핑
            torch.nn.utils.clip_grad_norm_(self.policy.parameters(), 1.0)
            
            self.optimizer.step()
            
            total_loss += loss.item()
        
        return total_loss / len(loader)
    
    def validate(self, loader):
        """검증"""
        total_loss = 0
        
        with torch.no_grad():
            for states, actions in loader:
                states = states.to(self.device)
                actions = actions.to(self.device)
                
                pred_actions = self.policy(states)
                loss = self.criterion(pred_actions, actions)
                
                total_loss += loss.item()
        
        return total_loss / len(loader)
```

### 4.1 정책 평가

```python
import mujoco
import numpy as np
import matplotlib.pyplot as plt

class BCEvaluator:
    def __init__(self, model_path, policy, dataset):
        self.model = mujoco.MjModel.from_xml_path(model_path)
        self.data = mujoco.MjData(self.model)
        self.policy = policy
        self.dataset = dataset
    
    def evaluate(self, num_episodes=10, render=False):
        """정책 평가"""
        results = {
            'success': 0,
            'rewards': [],
            'trajectories': [],
            'errors': []
        }
        
        for episode in range(num_episodes):
            # 초기화
            self.reset()
            
            episode_reward = 0
            trajectory = []
            
            for step in range(500):  # 최대 500 스텝
                # 상태 관찰
                state = self.get_state()
                
                # 정규화
                state_norm = (state - self.dataset.state_mean) / self.dataset.state_std
                
                # 행동 예측
                action_norm = self.policy.predict(state_norm)
                action = self.dataset.denormalize_action(action_norm)
                
                # 환경 스텝
                self.data.ctrl[:self.model.nu] = action
                mujoco.mj_step(self.model, self.data)
                
                # 보상 계산
                reward = self.compute_reward()
                episode_reward += reward
                
                # 궤적 저장
                trajectory.append({
                    'state': state,
                    'action': action,
                    'reward': reward
                })
            
            results['rewards'].append(episode_reward)
            results['trajectories'].append(trajectory)
            
            # 성공 판정
            if episode_reward > 100:  # 임의의 기준
                results['success'] += 1
        
        results['success_rate'] = results['success'] / num_episodes
        results['mean_reward'] = np.mean(results['rewards'])
        
        return results
    
    def reset(self):
        """환경 초기화"""
        for i in range(self.model.njnt):
            self.data.qpos[i] = 0
        self.data.qvel[:] = 0
    
    def get_state(self):
        """현재 상태 반환"""
        return np.concatenate([self.data.qpos, self.data.qvel])
    
    def compute_reward(self):
        """보상 계산"""
        target = np.array([0.3, 0.0, 0.5])
        current = self.data.site_xpos[0]
        
        distance = np.linalg.norm(target - current)
        reward = -distance  # 거리 기반 보상
        
        return reward
    
    def visualize_comparison(self, expert_states, bc_states):
        """Expert vs BC 비교 시각화"""
        fig, axes = plt.subplots(2, 2, figsize=(12, 8))
        
        # 위치 비교
        axes[0, 0].plot(expert_states[:, 0], label='Expert')
        axes[0, 0].plot(bc_states[:, 0], label='BC')
        axes[0, 0].set_title('X Position')
        axes[0, 0].legend()
        
        axes[0, 1].plot(expert_states[:, 1], label='Expert')
        axes[0, 1].plot(bc_states[:, 1], label='BC')
        axes[0, 1].set_title('Y Position')
        axes[0, 1].legend()
        
        # 속도 비교
        axes[1, 0].plot(expert_states[:, -3], label='Expert')
        axes[1, 0].plot(bc_states[:, -3], label='BC')
        axes[1, 0].set_title('X Velocity')
        axes[1, 0].legend()
        
        # 오차 분포
        errors = np.linalg.norm(expert_states - bc_states, axis=1)
        axes[1, 1].hist(errors, bins=50)
        axes[1, 1].set_title('Position Error Distribution')
        axes[1, 1].set_xlabel('Error (m)')
        
        plt.tight_layout()
        plt.savefig('bc_comparison.png')
        plt.show()
```

## 5. DAgger 구현

### 5.1 DAgger 알고리즘

```python
import torch
import numpy as np
from torch.utils.data import DataLoader, TensorDataset

class DAgger:
    def __init__(self, model_path, expert_policy, state_dim, action_dim, device='cuda'):
        self.model_path = model_path
        self.expert = expert_policy
        self.state_dim = state_dim
        self.action_dim = action_dim
        self.device = device
        
        # 데이터 버퍼
        self.states_buffer = []
        self.actions_buffer = []
        
        # 정책
        self.policy = BehaviorCloningPolicy(state_dim, action_dim).to(device)
        self.optimizer = torch.optim.Adam(self.policy.parameters(), lr=1e-3)
        self.criterion = torch.nn.MSELoss()
    
    def run_dagger(self, num_iterations=10, num_episodes=5, steps_per_episode=200, beta=0.5):
        """
        DAgger 알고리즘 실행
        
        Args:
            num_iterations: 반복 횟수
            num_episodes: 각 반복의 에피소드 수
            steps_per_episode: 에피소드당 스텝 수
            beta: Expert 비율 (0~1)
        """
        for iteration in range(num_iterations):
            print(f"\n=== DAgger Iteration {iteration + 1}/{num_iterations} ===")
            
            # 1. 현재 정책으로 환경 실행
            new_states, new_actions = self.collect_data(
                num_episodes, steps_per_episode, beta
            )
            
            # 2. 새 데이터 추가
            self.states_buffer.extend(new_states)
            self.actions_buffer.extend(new_actions)
            
            print(f"Total data points: {len(self.states_buffer)}")
            
            # 3. 정책 학습
            self.train_policy()
            
            # 4. 성능 평가
            if (iteration + 1) % 5 == 0:
                self.evaluate()
    
    def collect_data(self, num_episodes, steps_per_episode, beta):
        """데이터 수집"""
        model = mujoco.MjModel.from_xml_path(self.model_path)
        data = mujoco.MjData(model)
        
        new_states = []
        new_actions = []
        
        for episode in range(num_episodes):
            # 초기화
            self.reset_environment(model, data)
            
            for step in range(steps_per_episode):
                # 현재 상태
                state = self.get_state(model, data)
                
                # Expert 행동
                expert_action = self.expert.get_action(state)
                
                # 현재 정책 행동
                with torch.no_grad():
                    state_tensor = torch.FloatTensor(state).to(self.device)
                    policy_action = self.policy(state_tensor).cpu().numpy()
                
                # 혼합 행동 (Expert + Policy)
                if np.random.random() < beta:
                    action = expert_action  # Expert 사용
                    is_expert = True
                else:
                    action = policy_action  # 정책 사용
                    is_expert = False
                
                # 데이터 저장 (항상 expert 라벨 저장)
                new_states.append(state)
                new_actions.append(expert_action)  # expert 라벨
                
                # 환경 스텝
                data.ctrl[:model.nu] = action
                mujoco.mj_step(model, data)
        
        return new_states, new_actions
    
    def train_policy(self, num_epochs=50, batch_size=64):
        """정책 학습"""
        # 텐서 변환
        states = torch.FloatTensor(np.array(self.states_buffer)).to(self.device)
        actions = torch.FloatTensor(np.array(self.actions_buffer)).to(self.device)
        
        dataset = TensorDataset(states, actions)
        loader = DataLoader(dataset, batch_size=batch_size, shuffle=True)
        
        self.policy.train()
        
        for epoch in range(num_epochs):
            total_loss = 0
            
            for batch_states, batch_actions in loader:
                # 예측
                pred_actions = self.policy(batch_states)
                
                # 손실
                loss = self.criterion(pred_actions, batch_actions)
                
                # 역전파
                self.optimizer.zero_grad()
                loss.backward()
                self.optimizer.step()
                
                total_loss += loss.item()
            
            if (epoch + 1) % 10 == 0:
                print(f"  Epoch {epoch+1}/{num_epochs}, Loss: {total_loss/len(loader):.6f}")
    
    def evaluate(self, num_episodes=5):
        """정책 평가"""
        model = mujoco.MjModel.from_xml_path(self.model_path)
        data = mujoco.MjData(model)
        
        rewards = []
        
        for episode in range(num_episodes):
            self.reset_environment(model, data)
            
            episode_reward = 0
            
            for step in range(200):
                state = self.get_state(model, data)
                
                with torch.no_grad():
                    state_tensor = torch.FloatTensor(state).to(self.device)
                    action = self.policy(state_tensor).cpu().numpy()
                
                data.ctrl[:model.nu] = action
                mujoco.mj_step(model, data)
                
                reward = self.compute_reward(model, data)
                episode_reward += reward
            
            rewards.append(episode_reward)
        
        print(f"  Mean Reward: {np.mean(rewards):.2f} +/- {np.std(rewards):.2f}")
    
    def reset_environment(self, model, data):
        """환경 초기화"""
        for i in range(model.njnt):
            data.qpos[i] = np.random.uniform(-0.5, 0.5)
        data.qvel[:] = 0
    
    def get_state(self, model, data):
        """현재 상태"""
        return np.concatenate([data.qpos, data.qvel])
    
    def compute_reward(self, model, data):
        """보상 계산"""
        target = np.array([0.3, 0.0, 0.5])
        current = data.site_xpos[0]
        return -np.linalg.norm(target - current)
```
