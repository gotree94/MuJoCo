### 1.1 Gymnasium 인터페이스

```python
import gymnasium as gym
from gymnasium import spaces
import mujoco
import numpy as np

class MuJoCoRobotEnv(gym.Env):
    def __init__(self, model_path, max_steps=1000):
        super().__init__()
        
        self.model = mujoco.MjModel.from_xml_path(model_path)
        self.data = mujoco.MjData(self.model)
        
        self.observation_space = spaces.Box(
            low=-np.inf, high=np.inf,
            shape=(self.model.nq + self.model.nv,),
            dtype=np.float64
        )
        
        self.action_space = spaces.Box(
            low=-1.0, high=1.0,
            shape=(self.model.nu,),
            dtype=np.float64
        )
        
        self.max_steps = max_steps
        self.current_step = 0
        self.target_pos = np.array([0.3, 0.0, 0.5])
    
    def reset(self, seed=None, options=None):
        super().reset(seed=seed)
        
        for i in range(self.model.njnt):
            jr = self.model.jnt_range[i]
            self.data.qpos[i] = self.np_random.uniform(jr[0], jr[1])
        self.data.qvel[:] = 0
        
        self.current_step = 0
        return self._get_obs(), {}
    
    def step(self, action):
        self.data.ctrl[:] = action
        mujoco.mj_step(self.model, self.data)
        
        obs = self._get_obs()
        reward = self._compute_reward()
        terminated = self._check_terminated()
        truncated = self.current_step >= self.max_steps
        
        self.current_step += 1
        return obs, reward, terminated, truncated, {}
    
    def _get_obs(self):
        return np.concatenate([self.data.qpos, self.data.qvel])
    
    def _compute_reward(self):
        ee_pos = self.data.site_xpos[0]
        distance = np.linalg.norm(self.target_pos - ee_pos)
        return -distance - 0.001 * np.sum(self.data.ctrl**2)
    
    def _check_terminated(self):
        ee_pos = self.data.site_xpos[0]
        return np.linalg.norm(self.target_pos - ee_pos) < 0.02
```

## 2. PPO 알고리즘 구현

### 2.1 PPO 핵심 개념

```
PPO (Proximal Policy Optimization):
- 클리핑 objective로 안정적 학습
- advantage 추정 (GAE)
- 정책과 가치 함수 동시 학습

클리핑:
L_CLIP = E[min(r_t * A_t, clip(r_t, 1-eps, 1+eps) * A_t)]

r_t = pi_new(a|s) / pi_old(a|s)
```

### 2.2 신경망 구조

```python
import torch
import torch.nn as nn
import torch.nn.functional as F

class ActorCritic(nn.Module):
    def __init__(self, state_dim, action_dim, hidden_dim=256):
        super().__init__()
        
        # 공유 기반 네트워크
        self.shared = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU()
        )
        
        # Actor (정책)
        self.actor_mean = nn.Linear(hidden_dim, action_dim)
        self.actor_log_std = nn.Parameter(torch.zeros(action_dim))
        
        # Critic (가치 함수)
        self.critic = nn.Linear(hidden_dim, 1)
    
    def forward(self, state):
        shared_out = self.shared(state)
        
        action_mean = self.actor_mean(shared_out)
        action_std = self.actor_log_std.exp().expand_as(action_mean)
        
        value = self.critic(shared_out)
        
        return action_mean, action_std, value
    
    def get_action(self, state):
        action_mean, action_std, value = self.forward(state)
        
        dist = torch.distributions.Normal(action_mean, action_std)
        action = dist.sample()
        
        log_prob = dist.log_prob(action).sum(dim=-1)
        
        return action, log_prob, value.squeeze(-1)
    
    def evaluate(self, state, action):
        action_mean, action_std, value = self.forward(state)
        
        dist = torch.distributions.Normal(action_mean, action_std)
        log_prob = dist.log_prob(action).sum(dim=-1)
        entropy = dist.entropy().sum(dim=-1)
        
        return log_prob, entropy, value.squeeze(-1)
```

### 2.3 PPO 학습

```python
import numpy as np
from collections import deque

class PPOTrainer:
    def __init__(self, env, policy, device='cuda',
                 lr=3e-4, gamma=0.99, gae_lambda=0.95,
                 clip_epsilon=0.2, entropy_coef=0.01,
                 value_coef=0.5, max_grad_norm=0.5,
                 n_steps=2048, n_epochs=10, batch_size=64):
        
        self.env = env
        self.policy = policy.to(device)
        self.device = device
        
        self.gamma = gamma
        self.gae_lambda = gae_lambda
        self.clip_epsilon = clip_epsilon
        self.entropy_coef = entropy_coef
        self.value_coef = value_coef
        self.max_grad_norm = max_grad_norm
        self.n_steps = n_steps
        self.n_epochs = n_epochs
        self.batch_size = batch_size
        
        self.optimizer = torch.optim.Adam(policy.parameters(), lr=lr)
    
    def compute_gae(self, rewards, values, dones, next_value):
        """Generalized Advantage Estimation"""
        advantages = []
        gae = 0
        
        for t in reversed(range(len(rewards))):
            if t == len(rewards) - 1:
                next_val = next_value
            else:
                next_val = values[t + 1]
            
            delta = rewards[t] + self.gamma * next_val * (1 - dones[t]) - values[t]
            gae = delta + self.gamma * self.gae_lambda * (1 - dones[t]) * gae
            advantages.insert(0, gae)
        
        advantages = torch.FloatTensor(advantages).to(self.device)
        returns = advantages + torch.FloatTensor(values).to(self.device)
        
        return advantages, returns
    
    def collect_rollouts(self):
        """데이터 수집"""
        states = []
        actions = []
        log_probs = []
        rewards = []
        dones = []
        values = []
        
        state, _ = self.env.reset()
        
        for _ in range(self.n_steps):
            state_tensor = torch.FloatTensor(state).unsqueeze(0).to(self.device)
            
            with torch.no_grad():
                action, log_prob, value = self.policy.get_action(state_tensor)
            
            action_np = action.cpu().numpy().flatten()
            
            next_state, reward, terminated, truncated, _ = self.env.step(action_np)
            done = terminated or truncated
            
            states.append(state)
            actions.append(action.cpu().numpy().flatten())
            log_probs.append(log_prob.cpu().item())
            rewards.append(reward)
            dones.append(done)
            values.append(value.cpu().item())
            
            state = next_state
            
            if done:
                state, _ = self.env.reset()
        
        # 다음 상태의 가치
        with torch.no_grad():
            next_state_tensor = torch.FloatTensor(state).unsqueeze(0).to(self.device)
            _, _, next_value = self.policy.get_action(next_state_tensor)
            next_value = next_value.cpu().item()
        
        advantages, returns = self.compute_gae(rewards, values, dones, next_value)
        
        return {
            'states': np.array(states),
            'actions': np.array(actions),
            'log_probs': np.array(log_probs),
            'advantages': advantages,
            'returns': returns
        }
    
    def update(self, rollouts):
        """정책 업데이트"""
        states = torch.FloatTensor(rollouts['states']).to(self.device)
        actions = torch.FloatTensor(rollouts['actions']).to(self.device)
        old_log_probs = torch.FloatTensor(rollouts['log_probs']).to(self.device)
        advantages = rollouts['advantages']
        returns = rollouts['returns']
        
        # Advantage 정규화
        advantages = (advantages - advantages.mean()) / (advantages.std() + 1e-8)
        
        total_loss_history = []
        
        for _ in range(self.n_epochs):
            # 미니배치 학습
            indices = np.random.permutation(len(states))
            
            for start in range(0, len(states), self.batch_size):
                end = start + self.batch_size
                batch_indices = indices[start:end]
                
                batch_states = states[batch_indices]
                batch_actions = actions[batch_indices]
                batch_old_log_probs = old_log_probs[batch_indices]
                batch_advantages = advantages[batch_indices]
                batch_returns = returns[batch_indices]
                
                # 현재 정책 평가
                new_log_probs, entropy, values = self.policy.evaluate(
                    batch_states, batch_actions
                )
                
                # PPO 클리핑
                ratio = torch.exp(new_log_probs - batch_old_log_probs)
                surr1 = ratio * batch_advantages
                surr2 = torch.clamp(
                    ratio, 1 - self.clip_epsilon, 1 + self.clip_epsilon
                ) * batch_advantages
                
                actor_loss = -torch.min(surr1, surr2).mean()
                critic_loss = F.mse_loss(values, batch_returns)
                entropy_loss = -entropy.mean()
                
                total_loss = (actor_loss +
                             self.value_coef * critic_loss +
                             self.entropy_coef * entropy_loss)
                
                self.optimizer.zero_grad()
                total_loss.backward()
                
                torch.nn.utils.clip_grad_norm_(
                    self.policy.parameters(), self.max_grad_norm
                )
                
                self.optimizer.step()
                
                total_loss_history.append(total_loss.item())
        
        return np.mean(total_loss_history)
    
    def train(self, num_total_steps=100000, log_interval=10):
        """메인 학습 루프"""
        num_updates = num_total_steps // self.n_steps
        
        reward_history = []
        
        for update in range(num_updates):
            rollouts = self.collect_rollouts()
            
            loss = self.update(rollouts)
            
            # 평균 보상 계산
            mean_reward = np.sum(rollouts['returns']) / (self.n_steps)
            reward_history.append(mean_reward)
            
            if (update + 1) % log_interval == 0:
                print(f"Update {update+1}/{num_updates}")
                print(f"  Loss: {loss:.4f}")
                print(f"  Mean Reward: {mean_reward:.2f}")
        
        return reward_history
```

## 3. SAC 알고리즘 구현

### 3.1 SAC 핵심 개념

```
SAC (Soft Actor-Critic):
- Maximum Entropy RL
- 자동 온도 파라미터 조절
- 두 개의 Q-function (Mesof-Hyang trick)

목표 함수:
J(pi) = E[sum_t gamma^t (r_t + alpha * H(pi(.|s_t)))]

H: 엔트로피
alpha: 온도 파라미터
```

### 3.2 SAC 구현

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.distributions import Normal
import numpy as np

LOG_SIG_MAX = 2
LOG_SIG_MIN = -20
EPSILON = 1e-6

class SACActor(nn.Module):
    def __init__(self, state_dim, action_dim, hidden_dim=256):
        super().__init__()
        
        self.net = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU()
        )
        
        self.mean = nn.Linear(hidden_dim, action_dim)
        self.log_std = nn.Linear(hidden_dim, action_dim)
    
    def forward(self, state):
        x = self.net(state)
        mean = self.mean(x)
        log_std = self.log_std(x)
        log_std = torch.clamp(log_std, LOG_SIG_MIN, LOG_SIG_MAX)
        return mean, log_std
    
    def sample(self, state):
        mean, log_std = self.forward(state)
        std = log_std.exp()
        
        normal = Normal(mean, std)
        x_t = normal.rsample()
        action = torch.tanh(x_t)
        
        log_prob = normal.log_prob(x_t)
        log_prob -= torch.log(1 - action.pow(2) + EPSILON)
        log_prob = log_prob.sum(1, keepdim=True)
        
        return action, log_prob, mean

class SACCritic(nn.Module):
    def __init__(self, state_dim, action_dim, hidden_dim=256):
        super().__init__()
        
        self.q1 = nn.Sequential(
            nn.Linear(state_dim + action_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )
        
        self.q2 = nn.Sequential(
            nn.Linear(state_dim + action_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )
    
    def forward(self, state, action):
        sa = torch.cat([state, action], dim=1)
        return self.q1(sa), self.q2(sa)

class SACTrainer:
    def __init__(self, env, device='cuda',
                 lr=3e-4, gamma=0.99, tau=0.005,
                 alpha=0.2, auto_alpha=True, target_entropy=None):
        
        self.env = env
        self.device = device
        self.gamma = gamma
        self.tau = tau
        self.auto_alpha = auto_alpha
        
        state_dim = env.observation_space.shape[0]
        action_dim = env.action_space.shape[0]
        
        self.actor = SACActor(state_dim, action_dim).to(device)
        self.critic = SACCritic(state_dim, action_dim).to(device)
        self.critic_target = SACCritic(state_dim, action_dim).to(device)
        self.critic_target.load_state_dict(self.critic.state_dict())
        
        self.actor_optimizer = torch.optim.Adam(self.actor.parameters(), lr=lr)
        self.critic_optimizer = torch.optim.Adam(self.critic.parameters(), lr=lr)
        
        if auto_alpha:
            if target_entropy is None:
                self.target_entropy = -action_dim
            else:
                self.target_entropy = target_entropy
            self.log_alpha = torch.zeros(1, requires_grad=True, device=device)
            self.alpha_optimizer = torch.optim.Adam([self.log_alpha], lr=lr)
        else:
            self.alpha = alpha
    
    def select_action(self, state, evaluate=False):
        state_tensor = torch.FloatTensor(state).unsqueeze(0).to(self.device)
        
        if evaluate:
            _, _, mean = self.actor.sample(state_tensor)
            return mean.detach().cpu().numpy().flatten()
        else:
            action, _, _ = self.actor.sample(state_tensor)
            return action.detach().cpu().numpy().flatten()
    
    def update(self, batch):
        states, actions, rewards, next_states, dones = batch
        
        states = torch.FloatTensor(states).to(self.device)
        actions = torch.FloatTensor(actions).to(self.device)
        rewards = torch.FloatTensor(rewards).unsqueeze(1).to(self.device)
        next_states = torch.FloatTensor(next_states).to(self.device)
        dones = torch.FloatTensor(dones).unsqueeze(1).to(self.device)
        
        # Alpha 업데이트
        if self.auto_alpha:
            with torch.no_grad():
                new_actions, log_probs, _ = self.actor.sample(states)
            alpha_loss = -(self.log_alpha * (log_probs + self.target_entropy)).mean()
            
            self.alpha_optimizer.zero_grad()
            alpha_loss.backward()
            self.alpha_optimizer.step()
            
            self.alpha = self.log_alpha.exp()
        else:
            alpha_loss = torch.tensor(0.0)
        
        # Critic 업데이트
        with torch.no_grad():
            next_actions, next_log_probs, _ = self.actor.sample(next_states)
            q1_next, q2_next = self.critic_target(next_states, next_actions)
            q_next = torch.min(q1_next, q2_next) - self.alpha * next_log_probs
            q_target = rewards + self.gamma * (1 - dones) * q_next
        
        q1, q2 = self.critic(states, actions)
        critic_loss = F.mse_loss(q1, q_target) + F.mse_loss(q2, q_target)
        
        self.critic_optimizer.zero_grad()
        critic_loss.backward()
        self.critic_optimizer.step()
        
        # Actor 업데이트
        new_actions, log_probs, _ = self.actor.sample(states)
        q1_new, q2_new = self.critic(states, new_actions)
        q_new = torch.min(q1_new, q2_new)
        
        actor_loss = (self.alpha * log_probs - q_new).mean()
        
        self.actor_optimizer.zero_grad()
        actor_loss.backward()
        self.actor_optimizer.step()
        
        # Target network 소프트 업데이트
        for param, target_param in zip(self.critic.parameters(),
                                       self.critic_target.parameters()):
            target_param.data.copy_(
                self.tau * param.data + (1 - self.tau) * target_param.data
            )
        
        return actor_loss.item(), critic_loss.item(), self.alpha.item()
    
    def train(self, num_steps=100000, batch_size=256, warmup_steps=1000):
        from collections import deque
        import random
        
        replay_buffer = deque(maxlen=100000)
        state, _ = self.env.reset()
        
        total_steps = 0
        episode_reward = 0
        rewards_history = []
        
        for step in range(num_steps):
            if step < warmup_steps:
                action = self.env.action_space.sample()
            else:
                action = self.select_action(state)
            
            next_state, reward, terminated, truncated, _ = self.env.step(action)
            done = terminated or truncated
            
            replay_buffer.append((state, action, reward, next_state, done))
            
            state = next_state
            episode_reward += reward
            total_steps += 1
            
            if done:
                rewards_history.append(episode_reward)
                state, _ = self.env.reset()
                episode_reward = 0
            
            if len(replay_buffer) > batch_size and step >= warmup_steps:
                batch = random.sample(replay_buffer, batch_size)
                batch = map(np.array, zip(*batch))
                
                actor_loss, critic_loss, alpha = self.update(batch)
            
            if (step + 1) % 1000 == 0:
                mean_reward = np.mean(rewards_history[-100:]) if rewards_history else 0
                print(f"Step {step+1}/{num_steps}")
                print(f"  Mean Reward (last 100): {mean_reward:.2f}")
        
        return rewards_history
```

## 4. Domain Randomization

### 4.1 물리 파라미터 Randomization

```python
import numpy as np

class DomainRandomizer:
    def __init__(self):
        self.ranges = {
            'mass': (0.5, 2.0),           # 질량 배율
            'friction': (0.5, 2.0),       # 마찰 계수 배율
            'damping': (0.5, 2.0),        # 감쇠 배율
            'gravity': (-0.1, 0.1),       # 중력 변화
            'motor_strength': (0.8, 1.2), # 모터 힘 배율
            'sensor_noise': (0.0, 0.05),  # 센서 노이즈
        }
    
    def randomize_model(self, model):
        """MuJoCo 모델 파라미터 randomize"""
        for body_id in range(model.nbody):
            # 질량 randomize
            mass_scale = np.random.uniform(*self.ranges['mass'])
            model.body_mass[body_id] *= mass_scale
        
        for geom_id in range(model.ngeom):
            # 마찰 randomize
            friction_scale = np.random.uniform(*self.ranges['friction'])
            model.geom_friction[geom_id] *= friction_scale
        
        for joint_id in range(model.njnt):
            # 감쇠 randomize
            damping_scale = np.random.uniform(*self.ranges['damping'])
            model.dof_damping[joint_id] *= damping_scale
        
        return model
    
    def add_sensor_noise(self, state, noise_level=None):
        """센서 노이즈 추가"""
        if noise_level is None:
            noise_level = np.random.uniform(*self.ranges['sensor_noise'])
        
        noise = np.random.normal(0, noise_level, state.shape)
        return state + noise
    
    def randomize_action(self, action):
        """액추레이터 변형"""
        strength = np.random.uniform(*self.ranges['motor_strength'])
        return action * strength

class RandomizedEnv:
    def __init__(self, base_env):
        self.env = base_env
        self.randomizer = DomainRandomizer()
    
    def reset(self, **kwargs):
        # 매 에피소드마다 randomize
        self.randomizer.randomize_model(self.env.model)
        state, info = self.env.reset(**kwargs)
        return self.randomizer.add_sensor_noise(state), info
    
    def step(self, action):
        # 행동 변형
        randomized_action = self.randomizer.randomize_action(action)
        next_state, reward, terminated, truncated, info = self.env.step(randomized_action)
        
        # 센서 노이즈
        next_state = self.randomizer.add_sensor_noise(next_state)
        
        return next_state, reward, terminated, truncated, info
```
